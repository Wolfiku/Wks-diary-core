# wks-diary-core backend .env. Rename to .env, keep it off git entirely.

# Required to reach the API at all (header X-API-KEY). Generate: openssl rand -hex 32
WKS_API_KEY=CHANGE_ME_TO_A_LONG_RANDOM_SECRET

# Same 32-byte key your local Python client uses to encrypt/decrypt.
# The backend needs this to perform server-side merges and to validate restores.
# Generate: openssl rand -hex 32
WKS_VAULT_KEY=CHANGE_ME_TO_32_RANDOM_BYTES_AS_HEX

# Where encrypted blobs + history + the version log live
STORAGE_DIR=./storage

# Max upload size in bytes (default 50 MB)
MAX_UPLOAD_BYTES=52428800

# Address the server listens on
BIND_ADDR=0.0.0.0:8080

# ---------------------------------------------------------------------
# WKS_API_KEY and WKS_VAULT_KEY are DIFFERENT secrets with different
# blast radii. Losing WKS_API_KEY lets someone push/pull/merge/restore
# junk. Losing WKS_VAULT_KEY is what actually exposes diary content,
# because the backend decrypts during merges. Only run this backend on
# a server you fully trust (your own VPS), not third-party shared hosting.
# ---------------------------------------------------------------------
