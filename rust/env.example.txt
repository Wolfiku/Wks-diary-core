# wks-diary-core backend .env. Rename to .env, keep it off git entirely,
# and chmod 600 it (owned by the dedicated service user, not root).

# Required to reach the API at all (header X-API-KEY). Generate: openssl rand -hex 32
WKS_API_KEY=CHANGE_ME_TO_A_LONG_RANDOM_SECRET

# --- Vault encryption key: pick ONE of the two modes below ---

# Mode A (simple, works unattended/headless): raw 32-byte key as hex.
# Generate: openssl rand -hex 32
WKS_VAULT_KEY=CHANGE_ME_TO_32_RANDOM_BYTES_AS_HEX

# Mode B (recommended, more secure at rest): omit WKS_VAULT_KEY above and
# instead set a salt here; the server will then prompt for a passphrase
# on every manual start (does NOT work with unattended systemd restarts
# unless you accept typing it in via `systemctl start` + a terminal, or
# you fall back to Mode A for auto-restart convenience).
# WKS_VAULT_SALT=CHANGE_ME_TO_32_RANDOM_HEX_CHARS

# Where encrypted blobs + history + the version log live
STORAGE_DIR=./storage

# Max upload size in bytes (default 50 MB)
MAX_UPLOAD_BYTES=52428800

# Address the server listens on. Refuses to start on a non-loopback
# address unless WKS_ALLOW_PUBLIC_BIND=yes -- always put a TLS reverse
# proxy (Caddy/Nginx) in front instead of binding publicly here.
BIND_ADDR=127.0.0.1:8080
WKS_ALLOW_PUBLIC_BIND=no

# History retention: blobs older than this many days get pruned down to
# one snapshot per calendar week (log metadata is kept forever either way).
RETENTION_DAYS=30

# Simple global rate limit for failed auth attempts (brute-force mitigation).
RATE_LIMIT_MAX_FAILURES=10
RATE_LIMIT_WINDOW_SECS=60

# ---------------------------------------------------------------------
# WKS_API_KEY and the vault key are DIFFERENT secrets with different
# blast radii. Losing WKS_API_KEY lets someone push/pull/merge/restore
# junk (rate-limited). Losing the vault key is what actually exposes
# diary content, because the backend decrypts during merges, restores,
# and push-time validation. Only run this backend on infrastructure you
# fully trust.
# ---------------------------------------------------------------------
